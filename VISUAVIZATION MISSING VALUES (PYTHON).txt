# Step 1: Import Libraries
import pandas as pd
import matplotlib.pyplot as plt
import seaborn as sns

# Step 2: Load Dataset
df = pd.read_csv("sales_data.csv")

# Display data
print(df)

# -------------------------------
# Missing Values Visualization
# -------------------------------
plt.figure()
sns.heatmap(df.isnull(), cbar=False, cmap="viridis")
plt.title("Missing Values Heatmap")
plt.show()

# USE DATA SET 




# Count missing values
print("\nMissing Values Count:")
print(df.isnull().sum())

# -------------------------------
# Outlier Detection (Boxplot)
# -------------------------------
plt.figure()
sns.boxplot(data=df, y="Sales")
plt.title("Boxplot for Sales (Outliers Detection)")
plt.show()

plt.figure()
sns.boxplot(data=df, y="Profit")
plt.title("Boxplot for Profit (Outliers Detection)")
plt.show()

# -------------------------------
# Scatter Plot (Outliers View)
# -------------------------------
plt.figure()
sns.scatterplot(data=df, x="Sales", y="Profit")
plt.title("Scatter Plot (Outliers Detection)")
plt.show()
