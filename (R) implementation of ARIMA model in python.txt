# Step 1: Install required package (run once)
install.packages("forecast")

# Step 2: Load libraries
library(forecast)
library(ggplot2)

# Step 3: Generate Sample Time Series Data
set.seed(0)
n <- 50
time <- 1:n

# Create synthetic data (trend + noise)
data <- 0.5 * time + rnorm(n, mean = 0, sd = 2)

# Convert to time series object
ts_data <- ts(data)

# Step 4: Plot the Data
plot(ts_data,
     main = "Generated Time Series Data",
     xlab = "Time",
     ylab = "Value",
     col = "blue")

# Step 5: Apply ARIMA Model
model <- arima(ts_data, order = c(1, 1, 1))

# Step 6: Print Summary
print(model)

# Step 7: Forecast Future Values
forecast_values <- forecast(model, h = 10)

cat("\nForecasted Values:\n")
print(forecast_values)

# Step 8: Plot Forecast
plot(forecast_values,
     main = "ARIMA Forecast",
     xlab = "Time",
     ylab = "Value")
